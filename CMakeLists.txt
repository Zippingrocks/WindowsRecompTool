cmake_minimum_required(VERSION 3.20)
project(WinRecomp VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(WINRECOMP_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
set(ZYDIS_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_DOXYGEN OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ZYDIS_FEATURE_ENCODER OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/zydis/CMakeLists.txt")
  message(FATAL_ERROR "Initialize the pinned decoder: git submodule update --init --recursive third_party/zydis")
endif()
add_subdirectory(third_party/zydis EXCLUDE_FROM_ALL)
# Native x64 helpers preserve the host FP environment and the guest's 80-bit x87 values.
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64|X86_64)$")
  message(FATAL_ERROR "The current native runtime requires an x64 host")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_fp_asm.py
  --out ${CMAKE_CURRENT_BINARY_DIR}/native RESULT_VARIABLE WR_FP_GEN)
if(NOT WR_FP_GEN EQUAL 0)
  message(FATAL_ERROR "Native FP helper generation failed")
endif()
if(MSVC)
  enable_language(ASM_MASM)
  add_library(winrecomp_native STATIC ${CMAKE_CURRENT_BINARY_DIR}/native/x87_helpers.asm)
else()
  enable_language(ASM)
  add_library(winrecomp_native STATIC ${CMAKE_CURRENT_BINARY_DIR}/native/x87_helpers.S)
endif()
set_target_properties(winrecomp_native PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(winrecomp_core src/pe.cpp src/x86.cpp src/cfg.cpp src/lift.cpp)
target_include_directories(winrecomp_core PUBLIC include)
target_link_libraries(winrecomp_core PUBLIC Zydis)
if(MSVC)
  target_compile_options(winrecomp_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(winrecomp_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(WINRECOMP_SANITIZERS AND NOT MSVC)
  target_compile_options(winrecomp_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(winrecomp_core PUBLIC -fsanitize=address,undefined)
endif()
add_library(winrecomp_guest src/runtime/process.cpp src/runtime/win32.cpp)
target_link_libraries(winrecomp_guest PUBLIC winrecomp_core winrecomp_native)
add_executable(winrecomp src/main.cpp)
target_link_libraries(winrecomp PRIVATE winrecomp_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(winrecomp_tests tests/core.cpp)
  target_link_libraries(winrecomp_tests PRIVATE winrecomp_core)
  add_test(NAME core COMMAND winrecomp_tests)
  add_executable(winrecomp_fp_faults tests/fp_faults.cpp)
  target_link_libraries(winrecomp_fp_faults PRIVATE winrecomp_core winrecomp_native)
  add_test(NAME fp_faults COMMAND winrecomp_fp_faults)
  add_test(NAME target_fp_selector COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/target_fp_selector.py)
  add_test(NAME nls_data COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_nls.py --check)
  find_package(Threads REQUIRED)
  add_executable(winrecomp_nls_tests tests/nls.cpp)
  target_link_libraries(winrecomp_nls_tests PRIVATE winrecomp_guest)
  add_test(NAME nls COMMAND winrecomp_nls_tests)
  add_executable(winrecomp_process_tests tests/process.cpp)
  target_link_libraries(winrecomp_process_tests PRIVATE winrecomp_guest)
  add_test(NAME process COMMAND winrecomp_process_tests)
  add_executable(winrecomp_runtime_tests tests/runtime.cpp)
  target_link_libraries(winrecomp_runtime_tests PRIVATE winrecomp_core Threads::Threads)
  add_test(NAME runtime COMMAND winrecomp_runtime_tests)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME frontend COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/frontend.py $<TARGET_FILE:winrecomp>)
  endif()
endif()
option(WINRECOMP_CONFORMANCE "Compile translated fixtures and compare with external Unicorn" OFF)
if(BUILD_TESTING AND WINRECOMP_CONFORMANCE)
  if(NOT Python3_Interpreter_FOUND)
    message(FATAL_ERROR "Conformance tests require Python 3 and the optional Unicorn package")
  endif()
  add_test(NAME callback_abi COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/abi_callbacks.py
    --tool $<TARGET_FILE:winrecomp> --cxx ${CMAKE_CXX_COMPILER} --out ${CMAKE_CURRENT_BINARY_DIR}/abi_callbacks)
  add_test(NAME program COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/program.py
    --tool $<TARGET_FILE:winrecomp> --cxx ${CMAKE_CXX_COMPILER} --out ${CMAKE_CURRENT_BINARY_DIR}/program)
  add_test(NAME controlflow COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/controlflow.py
    --tool $<TARGET_FILE:winrecomp> --cxx ${CMAKE_CXX_COMPILER} --out ${CMAKE_CURRENT_BINARY_DIR}/controlflow)
  add_test(NAME simd COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/floating.py
    --tool $<TARGET_FILE:winrecomp> --cxx ${CMAKE_CXX_COMPILER} --out ${CMAKE_CURRENT_BINARY_DIR}/simd
    --native-lib $<TARGET_FILE:winrecomp_native> --vectors 192 --simd)
  add_test(NAME floating COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/floating.py
    --tool $<TARGET_FILE:winrecomp> --cxx ${CMAKE_CXX_COMPILER} --out ${CMAKE_CURRENT_BINARY_DIR}/floating
    --native-lib $<TARGET_FILE:winrecomp_native> --vectors 192)
  add_test(NAME conformance COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/conformance.py
    --tool $<TARGET_FILE:winrecomp> --cxx ${CMAKE_CXX_COMPILER}
    --out ${CMAKE_CURRENT_BINARY_DIR}/conformance --vectors 256)
endif()
